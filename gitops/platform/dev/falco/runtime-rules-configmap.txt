apiVersion: v1
kind: ConfigMap
metadata:
  name: runtime-rules-configmap
  namespace: falco

data:
  msef-runtime-rules.yaml: |
    - rule: MSEF Outbound Curl
      desc: Detect outbound curl execution
      condition: >
        spawned_process and
        proc.name = curl
      output: >
        MSEF_OUTBOUND_CURL
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        cmd=%proc.cmdline
      priority: WARNING
      tags: [msef, runtime]

    - rule: MSEF Network Connect
      desc: Detect outbound network connections
      condition: >
        outbound and container
      output: >
        MSEF_NETWORK_CONNECT
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        fd=%fd.name
      priority: WARNING
      tags: [msef, runtime]

    - rule: MSEF Sensitive File Read
      desc: Detect sensitive file reads
      condition: >
        open_read and
        (
          fd.name startswith /etc/shadow or
          fd.name startswith /var/run/secrets/kubernetes.io/serviceaccount
        )
      output: >
        MSEF_SENSITIVE_FILE
        pod=%k8s.pod.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime]

    - rule: MSEF Root Filesystem Write
      desc: Detect writes to protected filesystem
      condition: >
        open_write and
        (
          fd.name startswith /bin or
          fd.name startswith /usr or
          fd.name startswith /root
        )
      output: >
        MSEF_ROOT_WRITE
        pod=%k8s.pod.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime]

    - rule: MSEF Unexpected Shell Execution
      desc: Detect interactive shell execution
      condition: >
        spawned_process and
        container and
        proc.name in (bash, sh, ash, dash, zsh)
      output: >
        MSEF_UNEXPECTED_SHELL
        pod=%k8s.pod.name
        cmd=%proc.cmdline
      priority: WARNING
      tags: [msef, runtime]
