apiVersion: v1
kind: ConfigMap
metadata:
  name: runtime-rules-configmap
  namespace: falco
data:
  msef-runtime-rules.yaml: |
    # ==============================================================================
    # Multi-Layer Security Evaluation Framework (MSEF)
    # Custom Falco Rules ConfigMap - MITRE ATT&CK for Containers Evaluation
    # ==============================================================================

    # 1. T1071.001 - Outbound C2 / Web Protocols
    - rule: MSEF Outbound Curl
      desc: Detect outbound curl execution or HTTP beaconing
      condition: >
        spawned_process and
        container and
        proc.name in (curl, wget)
      output: >
        MSEF_OUTBOUND_CURL
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        cmd=%proc.cmdline
      priority: WARNING
      tags: [msef, runtime, attack, c2, mitre_t1071]

    # 2. General Network Recon & Lateral Probing
    - rule: MSEF Network Connect
      desc: Detect unexpected outbound container network connections
      condition: >
        outbound and container
      output: >
        MSEF_NETWORK_CONNECT
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        fd=%fd.name
      priority: WARNING
      tags: [msef, runtime, attack, lateral_movement]

    # 3. T1552.007 - Container Credential Access
    - rule: MSEF Sensitive File Read
      desc: Detect sensitive credential or service account token harvesting
      condition: >
        open_read and
        container and
        (
          fd.name startswith /etc/shadow or
          fd.name startswith /var/run/secrets/kubernetes.io/serviceaccount
        )
      output: >
        MSEF_SENSITIVE_FILE
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime, attack, credential_access, mitre_t1552]

    # 4. Host Path & Protected Filesystem Modification
    - rule: MSEF Root Filesystem Write
      desc: Detect writes to protected root system directories
      condition: >
        open_write and
        container and
        (
          fd.name startswith /bin or
          fd.name startswith /usr or
          fd.name startswith /etc or
          fd.name startswith /root
        )
      output: >
        MSEF_ROOT_WRITE
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime, attack, defense_evasion]

    # 5. T1059.004 - Interactive Terminal Shell Spawning
    - rule: MSEF Unexpected Shell Execution
      desc: Detect interactive shell execution within a running container
      condition: >
        spawned_process and
        container and
        proc.name in (bash, sh, ash, dash, zsh)
      output: >
        MSEF_UNEXPECTED_SHELL
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        cmd=%proc.cmdline
      priority: WARNING
      tags: [msef, runtime, attack, execution, mitre_t1059]

    # 6. T1611 - Container Escape via Cgroup Notification Release Agent
    - rule: MSEF Cgroup Release Agent Write
      desc: Detect attempts to escape container by tampering with cgroup release_agent
      condition: >
        open_write and
        container and
        fd.name contains "release_agent"
      output: >
        MSEF_CGROUP_ESCAPE
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime, attack, privilege_escalation, container_escape, mitre_t1611]

    # 7. T1036 - Masquerading & Execution from Ephemeral /tmp Scratch Space
    - rule: MSEF Execution from /tmp Scratch Space
      desc: Detect binary dropped and executed out of /tmp or /dev/shm
      condition: >
        spawned_process and
        container and
        (
          proc.name startswith "/tmp/" or
          proc.exepath startswith "/tmp/" or
          proc.name startswith "/dev/shm/" or
          proc.exepath startswith "/dev/shm/"
        )
      output: >
        MSEF_TMP_EXECUTION
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        cmd=%proc.cmdline
        exepath=%proc.exepath
      priority: WARNING
      tags: [msef, runtime, attack, defense_evasion, mitre_t1036]

    # 8. T1556 - Persistence via PAM or System Auth Tampering
    - rule: MSEF PAM Configuration Modification
      desc: Detect modification of sensitive authentication pam.d directory
      condition: >
        open_write and
        container and
        fd.name startswith "/etc/pam.d/"
      output: >
        MSEF_PAM_MODIFICATION
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        file=%fd.name
      priority: CRITICAL
      tags: [msef, runtime, attack, persistence, mitre_t1556]

    # 9. T1046 - Internal Cluster Service Discovery & Port Scanning
    - rule: MSEF Network Recon Scan
      desc: Detect internal port scanning against core Kubernetes services
      condition: >
        outbound and
        container and
        fd.sport > 1024 and
        fd.dport in (53, 80, 443, 6443, 2379, 8080, 9090)
      output: >
        MSEF_RECON_SCAN
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        dest=%fd.rip:%fd.dport
      priority: NOTICE
      tags: [msef, runtime, attack, lateral_movement, discovery, mitre_t1046]

    # 10. T1496 - Resource Hijacking & Crypto Mining Emulation
    - rule: MSEF Crypto Mining Outbound Beacon
      desc: Detect outbound communication to common stratum mining ports
      condition: >
        outbound and
        container and
        fd.dport in (3333, 5555, 7777, 9999)
      output: >
        MSEF_MINER_BEACON
        pod=%k8s.pod.name
        ns=%k8s.ns.name
        dest=%fd.rip:%fd.dport
      priority: CRITICAL
      tags: [msef, runtime, attack, impact, resource_hijacking, mitre_t1496]
